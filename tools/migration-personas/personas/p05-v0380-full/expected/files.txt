.slayzone/hooks/notify-default.sh  (text)
    #!/bin/sh
    # SlayZone agent lifecycle hook — BENIGN DUMB FORWARDER.
    #
    # SLAYZONE_NOTIFY_VERSION=4
    # ^ Bump on EVERY change to this file. The installer (notify-script-installer.ts)
    #   refuses to overwrite an on-disk script with a strictly-lower version, so a
    #   stale/older release channel can never downgrade a newer script on the
    #   shared ~/.slayzone/hooks/notify.sh. Newer is always backward-compatible.
    #
    # WHY THIS FILE IS LOGIC-FREE (the sustainable fix):
    #   ~/.slayzone/hooks/notify.sh is a SINGLE file shared by every release channel
    #   (getSlayzoneHomeDir() is NOT release-channel-scoped), and agents find it via
    #   the shared ~/.claude/settings.json pointer — so it CANNOT be
    #   release-channel-scoped. A shared file must therefore be BENIGN: it must
    #   contain no field-picking logic, so it does not matter which release channel's
    #   copy wins the last write. The previous version cherry-picked named fields
    #   (taskId, slaySessionId, session_id, hook_event_name, …) to forward; when an
    #   OLDER release channel's copy won, its field list was missing `slaySessionId`,
    #   so warm-pool sessions became invisible (no task resolution → no running
    #   spinner, no unread dot). This version forwards THREE opaque channels and lets
    #   the SERVER do all field extraction:
    #     - ctx : $SLAYZONE_AGENT_HOOK_CONTEXT — an opaque JSON blob the app packs at
    #             spawn with every identity field (taskId/slaySessionId/projectId/
    #             agentId/releaseChannel). Forwarded VERBATIM; never parsed here.
    #             v4 renamed it from $SLAYZONE_HOOK_CONTEXT (pairs with
    #             $SLAYZONE_AGENT_HOOK_URL); the old name is still read as a FALLBACK
    #             because this file is shared newest-wins across release channels, so
    #             an OLDER channel's app spawns agents that only set the old name.
    #             Dropping the fallback before every channel ships the new name would
    #             reproduce the v2→v3 clobber (empty ctx → no task resolution → no
    #             running spinner, no unread dot).
    #     - raw : the hook payload piped on stdin (Claude/Codex/Gemini/Antigravity).
    #     - arg : argv $1 — Antigravity passes the EVENT NAME here (its payload omits
    #             it); the OpenCode plugin passes the whole JSON payload here (no
    #             stdin). Forwarded opaquely as a JSON string; the server decides.
    #   Adding a new identity field later touches ONLY the app (writes ctx) and the
    #   server (reads ctx) — NEVER this script. That is the whole point.
    #
    # Installed by the app at ~/.slayzone/hooks/notify.sh (mode 0755).
    # Invoked by the host agent (e.g. Claude Code, Codex) via its hooks config.
    #
    # Required env (injected at PTY/chat spawn by buildMcpEnv):
    #   SLAYZONE_AGENT_HOOK_URL  - target URL, e.g. http://127.0.0.1:PORT/api/agent-hook
    #                              (a remote computer overlays its OWN loopback URL)
    #   SLAYZONE_AGENT_ID        - claude-code | codex | gemini | antigravity | opencode
    # Optional:
    #   SLAYZONE_AGENT_HOOK_CONTEXT - opaque identity blob (see above); sent as `ctx`
    #   SLAYZONE_HOOK_CONTEXT       - retired pre-v4 name; read only as a fallback
    #
    # Contract: ALWAYS exit 0. Hook failures must NOT bubble into the agent TUI
    # (Claude renders red error walls otherwise). Silent on any failure.
    
    set -e
    
    # Bail silently when not configured (e.g. agent run outside SlayZone).
    [ -z "$SLAYZONE_AGENT_HOOK_URL" ] && exit 0
    [ -z "$SLAYZONE_AGENT_ID" ] && exit 0
    
    # Gemini blocks waiting for a JSON response on stdout; empty {} = no-op.
    # Claude/Codex/Mastra/Droid discard our stdout, so this is universal.
    # Emit before payload read so it fires even if downstream POST fails.
    printf '{}\n'
    
    # The three opaque channels. NO grep/parse/named-field extraction here.
    #   ARG = argv $1 verbatim (may be an event name, a JSON payload, or empty).
    #   RAW = stdin verbatim (the hook payload, or empty).
    ARG="${1:-}"
    # Read stdin ONLY when argv $1 is NOT already the JSON payload. This is
    # load-bearing, not an optimization: the OpenCode plugin invokes us as
    # `bash notify.sh '<json>'` with NO stdin write, and inside a SlayZone PTY the
    # child inherits the terminal's stdin — which never sends EOF. An unconditional
    # `cat` would block FOREVER there, hanging the plugin's awaited exec (dark
    # badges — the exact failure this script exists to prevent). Every stdin-driven
    # agent (Claude/Codex/Gemini/Antigravity) writes its payload then CLOSES stdin,
    # so `cat` returns promptly for them. Mirrors the guard the pre-v2 script had.
    case "$ARG" in
      '{'*) RAW="" ;;
      *) RAW=$(cat 2>/dev/null || true) ;;
    esac
    
    # JSON-encode an arbitrary string into a quoted JSON string literal. This is the
    # ONLY logic left, and it is GENERIC string-escaping — NOT per-field naming, so
    # it cannot rot the way a field list does. Escapes backslash, double-quote, and
    # the whitespace control chars that occur in real hook data (\n \r \t). Other
    # C0 control bytes (\x00-\x1f) are not escaped — they never appear in an event
    # name or a JSON-stringified payload, the only two things that reach argv $1.
    # POSIX awk; falls back to a bare empty string on any failure.
    json_encode_string() {
      awk 'BEGIN {
        s = ARGV[1]
        out = "\""
        n = length(s)
        for (i = 1; i <= n; i++) {

    # SlayZone agent lifecycle hook — BENIGN DUMB FORWARDER.
    #
    # SLAYZONE_NOTIFY_VERSION=4
    # ^ Bump on EVERY change to this file. The installer (notify-script-installer.ts)
    #   refuses to overwrite an on-disk script with a strictly-lower version, so a
    #   stale/older release channel can never downgrade a newer script on the
    #   shared ~/.slayzone/hooks/notify.sh. Newer is always backward-compatible.
    #
    # WHY THIS FILE IS LOGIC-FREE (the sustainable fix):
    #   ~/.slayzone/hooks/notify.sh is a SINGLE file shared by every release channel
    #   (getSlayzoneHomeDir() is NOT release-channel-scoped), and agents find it via
    #   the shared ~/.claude/settings.json pointer — so it CANNOT be
    #   release-channel-scoped. A shared file must therefore be BENIGN: it must
    #   contain no field-picking logic, so it does not matter which release channel's
    #   copy wins the last write. The previous version cherry-picked named fields
    #   (taskId, slaySessionId, session_id, hook_event_name, …) to forward; when an
    #   OLDER release channel's copy won, its field list was missing `slaySessionId`,
    #   so warm-pool sessions became invisible (no task resolution → no running
    #   spinner, no unread dot). This version forwards THREE opaque channels and lets
    #   the SERVER do all field extraction:
    #     - ctx : $SLAYZONE_AGENT_HOOK_CONTEXT — an opaque JSON blob the app packs at
    #             spawn with every identity field (taskId/slaySessionId/projectId/
    #             agentId/releaseChannel). Forwarded VERBATIM; never parsed here.
    #             v4 renamed it from $SLAYZONE_HOOK_CONTEXT (pairs with
    #             $SLAYZONE_AGENT_HOOK_URL); the old name is still read as a FALLBACK
    #             because this file is shared newest-wins across release channels, so
    #             an OLDER channel's app spawns agents that only set the old name.
    #             Dropping the fallback before every channel ships the new name would
    #             reproduce the v2→v3 clobber (empty ctx → no task resolution → no
    #             running spinner, no unread dot).
    #     - raw : the hook payload piped on stdin (Claude/Codex/Gemini/Antigravity).
    #     - arg : argv $1 — Antigravity passes the EVENT NAME here (its payload omits
    #             it); the OpenCode plugin passes the whole JSON payload here (no
    #             stdin). Forwarded opaquely as a JSON string; the server decides.
    #   Adding a new identity field later touches ONLY the app (writes ctx) and the
    #   server (reads ctx) — NEVER this script. That is the whole point.
    #
    # Installed by the app at ~/.slayzone/hooks/notify.sh (mode 0755).
    # Invoked by the host agent (e.g. Claude Code, Codex) via its hooks config.
    #
    # Required env (injected at PTY/chat spawn by buildMcpEnv):
    #   SLAYZONE_AGENT_HOOK_URL  - target URL, e.g. http://127.0.0.1:PORT/api/agent-hook
    #                              (a remote computer overlays its OWN loopback URL)
    #   SLAYZONE_AGENT_ID        - claude-code | codex | gemini | antigravity | opencode
    # Optional:
    #   SLAYZONE_AGENT_HOOK_CONTEXT - opaque identity blob (see above); sent as `ctx`
    #   SLAYZONE_HOOK_CONTEXT       - retired pre-v4 name; read only as a fallback
    #
    # Contract: ALWAYS exit 0. Hook failures must NOT bubble into the agent TUI
    # (Claude renders red error walls otherwise). Silent on any failure.
    
    set -e
    
    # Bail silently when not configured (e.g. agent run outside SlayZone).
    [ -z "$SLAYZONE_AGENT_HOOK_URL" ] && exit 0
    [ -z "$SLAYZONE_AGENT_ID" ] && exit 0
    
    # Gemini blocks waiting for a JSON response on stdout; empty {} = no-op.
    # Claude/Codex/Mastra/Droid discard our stdout, so this is universal.
    # Emit before payload read so it fires even if downstream POST fails.
    printf '{}\n'
    
    # The three opaque channels. NO grep/parse/named-field extraction here.
    #   ARG = argv $1 verbatim (may be an event name, a JSON payload, or empty).
    #   RAW = stdin verbatim (the hook payload, or empty).
    ARG="${1:-}"
    # Read stdin ONLY when argv $1 is NOT already the JSON payload. This is
    # load-bearing, not an optimization: the OpenCode plugin invokes us as
    # `bash notify.sh '<json>'` with NO stdin write, and inside a SlayZone PTY the
    # child inherits the terminal's stdin — which never sends EOF. An unconditional
    # `cat` would block FOREVER there, hanging the plugin's awaited exec (dark
    # badges — the exact failure this script exists to prevent). Every stdin-driven
    # agent (Claude/Codex/Gemini/Antigravity) writes its payload then CLOSES stdin,
    # so `cat` returns promptly for them. Mirrors the guard the pre-v2 script had.
    case "$ARG" in
      '{'*) RAW="" ;;
      *) RAW=$(cat 2>/dev/null || true) ;;
    esac
    
    # JSON-encode an arbitrary string into a quoted JSON string literal. This is the
    # ONLY logic left, and it is GENERIC string-escaping — NOT per-field naming, so
    # it cannot rot the way a field list does. Escapes backslash, double-quote, and
    # the whitespace control chars that occur in real hook data (\n \r \t). Other
    # C0 control bytes (\x00-\x1f) are not escaped — they never appear in an event
    # name or a JSON-stringified payload, the only two things that reach argv $1.
    # POSIX awk; falls back to a bare empty string on any failure.
    json_encode_string() {
      awk 'BEGIN {
        s = ARGV[1]
        out = "\""
        n = length(s)
        for (i = 1; i <= n; i++) {
          c = substr(s, i, 1)
          if (c == "\\") out = out "\\\\"
          else if (c == "\"") out = out "\\\""
          else if (c == "\n") out = out "\\n"
          else if (c == "\r") out = out "\\r"
          else if (c == "\t") out = out "\\t"
          else out = out c
        }
        print out "\""
        exit
      }' "$1" 2>/dev/null || printf '""'
    }
    
    # ctx: forwarded VERBATIM (already valid JSON from the app), or `{}` when unset.
    # New name wins; the retired pre-v4 name is the fallback for an older release
    # channel's app spawning against this newer shared script (see header). Still NO
    # parsing — a name choice, not field-picking logic.
    CTX="${SLAYZONE_AGENT_HOOK_CONTEXT:-$SLAYZONE_HOOK_CONTEXT}"
    [ -z "$CTX" ] && CTX='{}'
    
    # raw: the stdin payload verbatim if it is present AND looks like JSON (starts
    # with '{'); otherwise JSON `null`. The server treats a null raw as "no stdin".
    case "$RAW" in
      '{'*) RAW_FIELD="$RAW" ;;
      *) RAW_FIELD="null" ;;
    esac
    
    # arg: argv $1 as a JSON string (or null when empty). Opaque — the server decides
    # whether it is an event name (Antigravity) or a JSON payload (OpenCode plugin).
    if [ -n "$ARG" ]; then
      ARG_FIELD=$(json_encode_string "$ARG")
    else
      ARG_FIELD="null"
    fi
    
    AGENT_ID_FIELD=$(json_encode_string "$SLAYZONE_AGENT_ID")
    
    ENVELOPE="{\"ctx\":$CTX,\"raw\":$RAW_FIELD,\"arg\":$ARG_FIELD,\"agentId\":$AGENT_ID_FIELD}"
    
    # Fire-and-forget. Errors swallowed; never block the agent.
    # curl is the primary path — present on macOS, most Linux, and bundled with
    # Git for Windows (whose bash runs this script). wget is a fallback for minimal
    # environments; if neither exists the POST is skipped silently. NO auth header:
    # the target is always loopback (local sidecar, or the computer's own loopback),
    # so no per-agent bearer is ever attached.
    if command -v curl >/dev/null 2>&1; then
      curl -s \
        --connect-timeout 2 \
        --max-time 5 \
        -X POST \
        -H 'Content-Type: application/json' \
        --data-binary "$ENVELOPE" \
        "$SLAYZONE_AGENT_HOOK_URL" \
        >/dev/null 2>&1 || true
    elif command -v wget >/dev/null 2>&1; then
      wget -q -O /dev/null \
        --timeout=5 \
        --header='Content-Type: application/json' \
        --post-data="$ENVELOPE" \
        "$SLAYZONE_AGENT_HOOK_URL" \
        >/dev/null 2>&1 || true
    fi
    
    exit 0

    // SlayZone opencode plugin v1
    //
    // Forwards OpenCode SDK session lifecycle events to the SlayZone agent-hook
    // notify script over a local shell exec. Plugin is loaded by OpenCode from
    //   ${XDG_CONFIG_HOME:-~/.config}/opencode/plugin/slayzone-notify.js
    //
    // The /home/persona/computers/c2/computer-runtime/hook-assets/0.44.0/notify.sh placeholder is substituted by the SlayZone installer
    // (packages/apps/app/src/main/agent-hooks/opencode-plugin-installer.ts) at
    // install time with the absolute path to `~/.slayzone/hooks/notify.sh`.
    //
    // Behavior:
    //   - No-op when SLAYZONE_TASK_ID is unset (plugin only fires inside SlayZone PTYs).
    //   - Subagent (child session) events are suppressed — oh-my-opencode + similar
    //     tools spawn many child sessions and would otherwise flap Start/Stop.
    //   - On client.session.list() error, sessions are assumed CHILD (safer to
    //     suppress than emit a false-positive Start/Stop).
    //   - Handles both modern (session.status w/ nested type) and legacy
    //     (session.busy / session.idle) event shapes for forward compat.
    //   - Singleton guard prevents double-load if OpenCode re-imports the plugin.
    
    export const SlayzoneNotifyPlugin = async ({ $, client }) => {
      if (globalThis.__slayzoneOpencodePluginV1) return {}
      globalThis.__slayzoneOpencodePluginV1 = true
      if (!process?.env?.SLAYZONE_TASK_ID) return {}
    
      const notifyPath = '/home/persona/computers/c2/computer-runtime/hook-assets/0.44.0/notify.sh'
      let currentState = 'idle' // 'idle' | 'busy'
      let rootSessionID = null // first non-child session we see
      let stopSent = false // dedup Stop emits
      const childSessionCache = new Map() // sessionID -> isChild bool
    
      const notify = async (eventName) => {
        const payload = JSON.stringify({ hook_event_name: eventName })
        try {
          await $`bash ${notifyPath} ${payload}`
        } catch {
          // Hook failures must never bubble into the OpenCode TUI.
        }
      }
    
      const isChildSession = async (sessionID) => {
        if (!sessionID) return true
        if (!client?.session?.list) return true
        if (childSessionCache.has(sessionID)) return childSessionCache.get(sessionID)
        try {
          const sessions = await client.session.list()
          const session = sessions.data?.find((s) => s.id === sessionID)
          const isChild = !!session?.parentID
          childSessionCache.set(sessionID, isChild)
          return isChild
        } catch {
          // On error, assume CHILD — safer than emitting a false-positive Start/Stop.
          return true
        }
      }
    
      const handleBusy = async (sessionID) => {
        if (!rootSessionID) rootSessionID = sessionID
        if (sessionID !== rootSessionID) return
        if (currentState === 'idle') {
          currentState = 'busy'
          stopSent = false
          await notify('Start')
        }
      }
    
      const handleStop = async (sessionID) => {
        if (rootSessionID && sessionID !== rootSessionID) return
        if (currentState === 'busy' && !stopSent) {
          currentState = 'idle'
          stopSent = true
          await notify('Stop')
          rootSessionID = null // reset for next session
        }
      }
    
      return {
        event: async ({ event }) => {
          const sessionID = event.properties?.sessionID ?? event.properties?.info?.id ?? null
    
          if (event.type === 'session.created') {
            const isChild = Boolean(event.properties?.info?.parentID)
            if (sessionID) childSessionCache.set(sessionID, isChild)
            if (!isChild) await notify('SessionStart')
            return
          }
    
          if (event.type === 'session.deleted') {
            const cached = sessionID ? childSessionCache.get(sessionID) : undefined
            const isChild = cached !== undefined ? cached : await isChildSession(sessionID)
            if (!isChild) await notify('SessionEnd')
            if (sessionID) childSessionCache.delete(sessionID)
            return
          }
    
          if (await isChildSession(sessionID)) return
    
          if (event.type === 'session.status') {
            const status = event.properties?.status
            if (status?.type === 'busy') await handleBusy(sessionID)
            else if (status?.type === 'idle') await handleStop(sessionID)
          }
          // Backwards-compat: some OpenCode versions emit busy/idle as top-level events.
          if (event.type === 'session.busy') await handleBusy(sessionID)
          if (event.type === 'session.idle') await handleStop(sessionID)
          if (event.type === 'session.error') await handleStop(sessionID)
        },
        'permission.ask': async (_permission, output) => {
          if (output.status === 'ask') await notify('PermissionRequest')
        }
      }
    }

    #!/usr/bin/env node
    const fs = require('node:fs'), path = require('node:path'), cp = require('node:child_process');
    const net = require('node:net'), os = require('node:os'), crypto = require('node:crypto');
    const root = process.env.SLAYZONE_ROOT;
    if (!root) throw Error('computer launcher requires SLAYZONE_ROOT');
    const role = process.argv.includes('--session-host') ? 'host' : 'supervisor';
    const dir = path.join(root, 'computer-runtime'), pointer = path.join(dir, role === 'host' ? 'host-release.json' : 'supervisor.json');
    // Restart delay after consecutive failures: 2s doubling to 60s, reset once a child is healthy.
    const BACKOFF_MIN_MS = 2000, BACKOFF_MAX_MS = 60000, LOG_MAX_BYTES = 5 * 1024 * 1024;
    // A session host that found another live host on its control endpoint (computer-endpoints.ts).
    const SUPERSEDED_EXIT = 79;
    // This root's endpoints, derived exactly as computer-endpoints.ts derives them.
    const endpointKey = crypto.createHash('sha256').update(path.resolve(root)).digest('hex').slice(0, 24);
    const endpointDir = process.platform === 'win32' ? null : path.join(os.tmpdir(), 'slay-computer-' + (process.getuid ? process.getuid() : 'user') + '-' + endpointKey);
    const endpoints = endpointDir
      ? {control: path.join(endpointDir, 'control.sock'), launcher: path.join(endpointDir, 'l.sock')}
      : {control: '\\\\.\\pipe\\slay-computer-' + endpointKey, launcher: '\\\\.\\pipe\\slay-computer-launcher-' + endpointKey};
    // Longest Unix socket path Node accepts (computer-endpoints.ts UNIX_SOCKET_PATH_MAX_BYTES):
    // a longer one fails listen/connect with a bare EINVAL, so it is refused by name first.
    const SOCKET_PATH_MAX = {"darwin":104,"linux":108}[process.platform] || 104;
    // listen errors meaning the endpoint path is already held (computer-endpoints.ts
    // ENDPOINT_IN_USE_LISTEN_ERRORS): EADDRINUSE, or EEXIST when macOS sees another
    // launcher bind the same path at the same moment. Either way: probe, never bail.
    const IN_USE_LISTEN_ERRORS = ["EADDRINUSE","EEXIST"];
    function socketPathError(p) {
      const bytes = Buffer.byteLength(p);
      return bytes <= SOCKET_PATH_MAX ? null : 'computer endpoint ' + p + ' is ' + bytes + ' bytes, over the ' + SOCKET_PATH_MAX + '-byte Unix socket path limit on ' + process.platform + '; point TMPDIR at a shorter directory';
    }
    if (process.argv.includes('--print-endpoints')) { process.stdout.write(JSON.stringify(endpoints) + '\n'); return; }
    let child, stopping = false, healthy = false, timer, probation, killTimer, failures = 0, nextTimer, lock;
    // What a waiting supervisor asks over the launcher endpoint (queryHostLauncher).
    const status = {state: 'starting', childPid: null, nextLaunchAt: null, lastExit: null};
    function read() { return JSON.parse(fs.readFileSync(pointer, 'utf8')); }
    // Every start and exit of the child, timestamped, to stderr and to logs/computer.log
    // (shared by every computer process). Before, an exit left no trace at all here.
    function log(msg) {
      const line = new Date().toISOString() + ' [launcher ' + process.pid + '] ' + role + ' ' + msg + '\n';
      try { process.stderr.write(line); } catch {}
      try { const logs = path.join(root, 'logs'); fs.mkdirSync(logs, {recursive: true, mode: 0o700}); fs.appendFileSync(path.join(logs, 'computer.log'), line, {mode: 0o600}); } catch {}
    }
    // The session host's launcher writes, for its whole life, to the host.log it was started
    // with: rotated only when a host was spawned, it grew without bound (32 MB). Copy, then
    // truncate in place: every holder of the file appends, so all keep writing to its start.
    function rotateHostLog() {
      if (role !== 'host') return;
      try {
        const file = path.join(dir, 'host.log'), out = fs.fstatSync(1), st = fs.statSync(file);
        if (out.ino !== st.ino || out.dev !== st.dev || st.size <= LOG_MAX_BYTES) return;
        fs.copyFileSync(file, file + '.1'); fs.truncateSync(file, 0);
      } catch {}
    }
    setInterval(rotateHostLog, 600000).unref();
    // ── One session-host launcher per root ──────────────────────────────────────────
    // The launcher endpoint is held for this launcher's whole life: a second launcher for
    // the root finds it live and stops before starting any child. (Two once ran at once,
    // one restarting an EADDRINUSE-failing child forever, rolling back good releases.)
    function prepareEndpointDir() {
      if (!endpointDir) return;
      try { fs.mkdirSync(endpointDir, {mode: 0o700}); } catch (e) { if (e.code !== 'EEXIST') throw e; }
      const st = fs.lstatSync(endpointDir);
      if (st.isSymbolicLink() || !st.isDirectory() || (process.getuid && st.uid !== process.getuid()) || (st.mode & 0o077))
        throw Error('unsafe computer endpoint directory ' + endpointDir + ': must be a directory owned by this user with mode 0700');
    }
    // live / stale / absent as probeEndpoint (computer-endpoints.ts) tells them; any other
    // connect error says nothing about an owner and is passed on (cb(null, err)).
    function probe(cb) {
      const s = net.connect(endpoints.launcher); let done = false;
      const fin = (r, err) => { if (done) return; done = true; clearTimeout(t); s.destroy(); cb(r, err); };
      const t = setTimeout(() => fin('live'), 2000);
      s.once('connect', () => fin('live'));
      s.once('error', (e) => e.code === 'ENOENT' ? fin('absent') : e.code === 'ECONNREFUSED' ? fin('stale') : e.code === 'EAGAIN' ? fin('live') : fin(null, e));
    }
    function serveStatus(socket) {
      let buf = '';
      socket.setEncoding('utf8'); socket.on('error', () => {}); socket.setTimeout(2000, () => socket.destroy());
      socket.on('data', (chunk) => {
        buf += chunk; const end = buf.indexOf('\n');
        if (end < 0) { if (buf.length > 1024) socket.destroy(); return; }
        let command; try { command = JSON.parse(buf.slice(0, end)).command; } catch {}
        if (command === 'launch-now' && status.state === 'backoff' && !stopping) { clearTimeout(nextTimer); log('launching now, at a waiting supervisor\'s request'); launch(); }
        socket.end(JSON.stringify(status) + '\n');
      });
    }
    function bail(msg) { log(msg + '; stopping'); process.exit(1); }
    // Every bind of the launcher endpoint happens inside one short critical section (an
    // atomic mkdir of the guard). listen() on a Unix socket is bind() then listen(): in
    // between, the file exists but refuses connections, exactly like one whose launcher
    // died. With binds serialized, a refusing socket seen while holding the guard can only
    // be a dead launcher's, so replacing it is safe. (2026-11-27: under full CPU load a
    // launcher probed a winner preempted in that gap, "took over" its live endpoint, and
    // two launchers each ran a host.) Windows: named pipes are exclusive; no guard needed.
    // The guard keeps its old name so an older launcher.cjs still running excludes with it.
    const GUARD_STALE_MS = 10000;
    function acquire(then, attempt) {
      attempt = attempt || 0;
      if (attempt > 50) return bail('cannot take the launcher endpoint ' + endpoints.launcher);
      if (endpointDir) { const tooLong = socketPathError(endpoints.launcher) || socketPathError(endpoints.control); if (tooLong) return bail(tooLong); }
      try { prepareEndpointDir(); } catch (e) { return bail(e.message); }
      const retry = (ms) => setTimeout(() => acquire(then, attempt + 1), ms);
      const guard = endpointDir ? endpoints.launcher + '.takeover' : null;
      if (guard) {
        try { fs.mkdirSync(guard, {mode: 0o700}); } catch (e) {
          // ENOENT: a launcher exiting just removed the (empty) endpoint directory.
          if (e.code === 'ENOENT') return retry(0);
          if (e.code !== 'EEXIST') return bail('cannot take ' + guard + ': ' + e.message);
          // Held for a bind (milliseconds). Waiting on it is not a failed attempt: it ends
          // when the holder finishes, or GUARD_STALE_MS after a holder that died holding it.
          try { if (Date.now() - fs.statSync(guard).mtimeMs > GUARD_STALE_MS) fs.rmdirSync(guard); } catch {}
          return setTimeout(() => acquire(then, attempt), 25);
        }
      }
      const unguard = () => { if (guard) try { fs.rmdirSync(guard); } catch {} };
      const won = (server, how) => { unguard(); if (how) log(how); lock = server; then(); };
      const server = net.createServer(serveStatus);
      server.once('error', (err) => {
        if (!IN_USE_LISTEN_ERRORS.includes(err.code)) { unguard(); return bail('cannot bind ' + endpoints.launcher + ': ' + err.message); }
        probe((state, probeErr) => {
          if (probeErr) { unguard(); return bail('cannot probe ' + endpoints.launcher + ': ' + (probeErr.code || probeErr.message)); }
          if (state === 'live') { unguard(); log('launcher already running for this root; superseded, stopping'); process.exit(0); }
          if (state === 'absent' || !guard) { unguard(); return retry(50); }
          // Stale: its launcher died (no live one binds without the guard this one holds).
          try { fs.unlinkSync(endpoints.launcher); } catch {}
          const taken = net.createServer(serveStatus);
          taken.once('error', () => { unguard(); retry(0); });
          taken.listen(endpoints.launcher, () => won(taken, 'took over a launcher endpoint left by a dead launcher'));
        });
      });
      server.listen(endpoints.launcher, () => won(server));
    }
    // Ends this launcher: frees its endpoint (and the directory, once empty), then exits.
    function release(code) {
      status.state = 'stopping'; status.nextLaunchAt = null; clearTimeout(nextTimer);
      process.exitCode = code;
      if (!lock) return;
      lock.close(() => { if (endpointDir) try { fs.rmdirSync(endpointDir); } catch {} });
      lock = undefined;
    }
    function launch() {
      if (stopping) return;
      rotateHostLog();
      const state = read(); let timedOut = false; healthy = false; probation = undefined;
      const startedAt = Date.now();
      child = cp.fork(state.current.entry, [role === 'host' ? '--session-host' : '--supervisor'], {env: {...process.env, SLAYZONE_NONINTERACTIVE:'1'}, stdio:['inherit','inherit','inherit','ipc']});
      status.state = 'running'; status.childPid = child.pid; status.nextLaunchAt = null;
      log(state.current.version + ' started pid=' + child.pid + (failures ? ' (after ' + failures + ' consecutive failures)' : ''));
      timer = setTimeout(() => {
        timedOut = true;
        log(state.current.version + ' startup timed out; restarting');
        // A supervisor SIGTERM means explicit shutdown, including its persistent host.
        // Crash only the supervisor; its connectors stop on IPC loss and work survives.
        if (role === 'supervisor') child.kill('SIGKILL');
        else { child.kill('SIGTERM'); killTimer = setTimeout(() => child.kill('SIGKILL'), 10000); }
      }, 90000);
      child.on('message', m => { if (m && m.type === 'supervisor-ready') {clearTimeout(timer); if (!probation) probation = setTimeout(() => {healthy = true; failures = 0;}, 60000);} });
      child.on('error', (err) => { log(state.current.version + ' child error: ' + (err && err.message)); });
      child.on('exit', (code, signal) => {
        clearTimeout(timer); clearTimeout(probation); clearTimeout(killTimer);

    #
    # SLAYZONE_NOTIFY_VERSION=4
    # ^ Bump on EVERY change to this file. The installer (notify-script-installer.ts)
    #   refuses to overwrite an on-disk script with a strictly-lower version, so a
    #   stale/older release channel can never downgrade a newer script on the
    #   shared ~/.slayzone/hooks/notify.sh. Newer is always backward-compatible.
    #
    # WHY THIS FILE IS LOGIC-FREE (the sustainable fix):
    #   ~/.slayzone/hooks/notify.sh is a SINGLE file shared by every release channel
    #   (getSlayzoneHomeDir() is NOT release-channel-scoped), and agents find it via
    #   the shared ~/.claude/settings.json pointer — so it CANNOT be
    #   release-channel-scoped. A shared file must therefore be BENIGN: it must
    #   contain no field-picking logic, so it does not matter which release channel's
    #   copy wins the last write. The previous version cherry-picked named fields
    #   (taskId, slaySessionId, session_id, hook_event_name, …) to forward; when an
    #   OLDER release channel's copy won, its field list was missing `slaySessionId`,
    #   so warm-pool sessions became invisible (no task resolution → no running
    #   spinner, no unread dot). This version forwards THREE opaque channels and lets
    #   the SERVER do all field extraction:
    #     - ctx : $SLAYZONE_AGENT_HOOK_CONTEXT — an opaque JSON blob the app packs at
    #             spawn with every identity field (taskId/slaySessionId/projectId/
    #             agentId/releaseChannel). Forwarded VERBATIM; never parsed here.
    #             v4 renamed it from $SLAYZONE_HOOK_CONTEXT (pairs with
    #             $SLAYZONE_AGENT_HOOK_URL); the old name is still read as a FALLBACK
    #             because this file is shared newest-wins across release channels, so
    #             an OLDER channel's app spawns agents that only set the old name.
    #             Dropping the fallback before every channel ships the new name would
    #             reproduce the v2→v3 clobber (empty ctx → no task resolution → no
    #             running spinner, no unread dot).
    #     - raw : the hook payload piped on stdin (Claude/Codex/Gemini/Antigravity).
    #     - arg : argv $1 — Antigravity passes the EVENT NAME here (its payload omits
    #             it); the OpenCode plugin passes the whole JSON payload here (no
    #             stdin). Forwarded opaquely as a JSON string; the server decides.
    #   Adding a new identity field later touches ONLY the app (writes ctx) and the
    #   server (reads ctx) — NEVER this script. That is the whole point.
    #
    # Installed by the app at ~/.slayzone/hooks/notify.sh (mode 0755).
    # Invoked by the host agent (e.g. Claude Code, Codex) via its hooks config.
    #
    # Required env (injected at PTY/chat spawn by buildMcpEnv):
    #   SLAYZONE_AGENT_HOOK_URL  - target URL, e.g. http://127.0.0.1:PORT/api/agent-hook
    #                              (a remote computer overlays its OWN loopback URL)
    #   SLAYZONE_AGENT_ID        - claude-code | codex | gemini | antigravity | opencode
    # Optional:
    #   SLAYZONE_AGENT_HOOK_CONTEXT - opaque identity blob (see above); sent as `ctx`
    #   SLAYZONE_HOOK_CONTEXT       - retired pre-v4 name; read only as a fallback
    #
    # Contract: ALWAYS exit 0. Hook failures must NOT bubble into the agent TUI
    # (Claude renders red error walls otherwise). Silent on any failure.
    
    set -e

    [ -z "$SLAYZONE_AGENT_HOOK_URL" ] && exit 0
    [ -z "$SLAYZONE_AGENT_ID" ] && exit 0
    
    # Gemini blocks waiting for a JSON response on stdout; empty {} = no-op.
    # Claude/Codex/Mastra/Droid discard our stdout, so this is universal.
    # Emit before payload read so it fires even if downstream POST fails.
    printf '{}\n'

    #   ARG = argv $1 verbatim (may be an event name, a JSON payload, or empty).
    #   RAW = stdin verbatim (the hook payload, or empty).
    ARG="${1:-}"

    # load-bearing, not an optimization: the OpenCode plugin invokes us as
    # `bash notify.sh '<json>'` with NO stdin write, and inside a SlayZone PTY the
    # child inherits the terminal's stdin — which never sends EOF. An unconditional
    # `cat` would block FOREVER there, hanging the plugin's awaited exec (dark
    # badges — the exact failure this script exists to prevent). Every stdin-driven
    # agent (Claude/Codex/Gemini/Antigravity) writes its payload then CLOSES stdin,
    # so `cat` returns promptly for them. Mirrors the guard the pre-v2 script had.
    case "$ARG" in
      '{'*) RAW="" ;;
      *) RAW=$(cat 2>/dev/null || true) ;;
    esac
    
    # JSON-encode an arbitrary string into a quoted JSON string literal. This is the
    # ONLY logic left, and it is GENERIC string-escaping — NOT per-field naming, so
    # it cannot rot the way a field list does. Escapes backslash, double-quote, and
    # the whitespace control chars that occur in real hook data (\n \r \t). Other
    # C0 control bytes (\x00-\x1f) are not escaped — they never appear in an event
    # name or a JSON-stringified payload, the only two things that reach argv $1.
    # POSIX awk; falls back to a bare empty string on any failure.
    json_encode_string() {
      awk 'BEGIN {
        s = ARGV[1]
        out = "\""
        n = length(s)
        for (i = 1; i <= n; i++) {
          c = substr(s, i, 1)
          if (c == "\\") out = out "\\\\"
          else if (c == "\"") out = out "\\\""
          else if (c == "\n") out = out "\\n"
          else if (c == "\r") out = out "\\r"
          else if (c == "\t") out = out "\\t"
          else out = out c
        }
        print out "\""
        exit
      }' "$1" 2>/dev/null || printf '""'
    }
    
    # ctx: forwarded VERBATIM (already valid JSON from the app), or `{}` when unset.
    # New name wins; the retired pre-v4 name is the fallback for an older release
    # channel's app spawning against this newer shared script (see header). Still NO
    # parsing — a name choice, not field-picking logic.
    CTX="${SLAYZONE_AGENT_HOOK_CONTEXT:-$SLAYZONE_HOOK_CONTEXT}"
    [ -z "$CTX" ] && CTX='{}'
    
    # raw: the stdin payload verbatim if it is present AND looks like JSON (starts
    # with '{'); otherwise JSON `null`. The server treats a null raw as "no stdin".
    case "$RAW" in
      '{'*) RAW_FIELD="$RAW" ;;
      *) RAW_FIELD="null" ;;
    esac
    
    # arg: argv $1 as a JSON string (or null when empty). Opaque — the server decides
    # whether it is an event name (Antigravity) or a JSON payload (OpenCode plugin).
    if [ -n "$ARG" ]; then
      ARG_FIELD=$(json_encode_string "$ARG")
    else
      ARG_FIELD="null"
    fi
    
    AGENT_ID_FIELD=$(json_encode_string "$SLAYZONE_AGENT_ID")
    
    ENVELOPE="{\"ctx\":$CTX,\"raw\":$RAW_FIELD,\"arg\":$ARG_FIELD,\"agentId\":$AGENT_ID_FIELD}"
    
    # Fire-and-forget. Errors swallowed; never block the agent.
    # curl is the primary path — present on macOS, most Linux, and bundled with
    # Git for Windows (whose bash runs this script). wget is a fallback for minimal
    # environments; if neither exists the POST is skipped silently. NO auth header:
    # the target is always loopback (local sidecar, or the computer's own loopback),
    # so no per-agent bearer is ever attached.
    if command -v curl >/dev/null 2>&1; then
      curl -s \
        --connect-timeout 2 \
        --max-time 5 \
        -X POST \
        -H 'Content-Type: application/json' \
        --data-binary "$ENVELOPE" \
        "$SLAYZONE_AGENT_HOOK_URL" \
        >/dev/null 2>&1 || true
    elif command -v wget >/dev/null 2>&1; then
      wget -q -O /dev/null \
        --timeout=5 \
        --header='Content-Type: application/json' \
        --post-data="$ENVELOPE" \
        "$SLAYZONE_AGENT_HOOK_URL" \
        >/dev/null 2>&1 || true
    fi
    
    exit 0

    // SlayZone opencode plugin v1
    //
    // Forwards OpenCode SDK session lifecycle events to the SlayZone agent-hook
    // notify script over a local shell exec. Plugin is loaded by OpenCode from
    //   ${XDG_CONFIG_HOME:-~/.config}/opencode/plugin/slayzone-notify.js
    //
    // The /home/persona/computers/c3/computer-runtime/hook-assets/0.38.0/notify.sh placeholder is substituted by the SlayZone installer
    // (packages/apps/app/src/main/agent-hooks/opencode-plugin-installer.ts) at
    // install time with the absolute path to `~/.slayzone/hooks/notify.sh`.
    //
    // Behavior:
    //   - No-op when SLAYZONE_TASK_ID is unset (plugin only fires inside SlayZone PTYs).
    //   - Subagent (child session) events are suppressed — oh-my-opencode + similar
    //     tools spawn many child sessions and would otherwise flap Start/Stop.
    //   - On client.session.list() error, sessions are assumed CHILD (safer to
    //     suppress than emit a false-positive Start/Stop).
    //   - Handles both modern (session.status w/ nested type) and legacy
    //     (session.busy / session.idle) event shapes for forward compat.
    //   - Singleton guard prevents double-load if OpenCode re-imports the plugin.
    
    export const SlayzoneNotifyPlugin = async ({ $, client }) => {
      if (globalThis.__slayzoneOpencodePluginV1) return {}
      globalThis.__slayzoneOpencodePluginV1 = true
      if (!process?.env?.SLAYZONE_TASK_ID) return {}
    
      const notifyPath = '/home/persona/computers/c3/computer-runtime/hook-assets/0.38.0/notify.sh'
      let currentState = 'idle' // 'idle' | 'busy'
      let rootSessionID = null // first non-child session we see
      let stopSent = false // dedup Stop emits
      const childSessionCache = new Map() // sessionID -> isChild bool
    
      const notify = async (eventName) => {
        const payload = JSON.stringify({ hook_event_name: eventName })
        try {
          await $`bash ${notifyPath} ${payload}`
        } catch {
          // Hook failures must never bubble into the OpenCode TUI.
        }
      }
    
      const isChildSession = async (sessionID) => {
        if (!sessionID) return true
        if (!client?.session?.list) return true
        if (childSessionCache.has(sessionID)) return childSessionCache.get(sessionID)
        try {
          const sessions = await client.session.list()
          const session = sessions.data?.find((s) => s.id === sessionID)
          const isChild = !!session?.parentID
          childSessionCache.set(sessionID, isChild)
          return isChild
        } catch {
          // On error, assume CHILD — safer than emitting a false-positive Start/Stop.
          return true
        }
      }
    
      const handleBusy = async (sessionID) => {
        if (!rootSessionID) rootSessionID = sessionID
        if (sessionID !== rootSessionID) return
        if (currentState === 'idle') {
          currentState = 'busy'
          stopSent = false
          await notify('Start')
        }
      }
    
      const handleStop = async (sessionID) => {
        if (rootSessionID && sessionID !== rootSessionID) return
        if (currentState === 'busy' && !stopSent) {
          currentState = 'idle'
          stopSent = true
          await notify('Stop')
          rootSessionID = null // reset for next session
        }
      }
    
      return {
        event: async ({ event }) => {
          const sessionID = event.properties?.sessionID ?? event.properties?.info?.id ?? null
    
          if (event.type === 'session.created') {
            const isChild = Boolean(event.properties?.info?.parentID)
            if (sessionID) childSessionCache.set(sessionID, isChild)
            if (!isChild) await notify('SessionStart')
            return
          }
    
          if (event.type === 'session.deleted') {
            const cached = sessionID ? childSessionCache.get(sessionID) : undefined
            const isChild = cached !== undefined ? cached : await isChildSession(sessionID)
            if (!isChild) await notify('SessionEnd')
            if (sessionID) childSessionCache.delete(sessionID)
            return
          }
    
          if (await isChildSession(sessionID)) return
    
          if (event.type === 'session.status') {
            const status = event.properties?.status
            if (status?.type === 'busy') await handleBusy(sessionID)
            else if (status?.type === 'idle') await handleStop(sessionID)
          }
          // Backwards-compat: some OpenCode versions emit busy/idle as top-level events.
          if (event.type === 'session.busy') await handleBusy(sessionID)
          if (event.type === 'session.idle') await handleStop(sessionID)
          if (event.type === 'session.error') await handleStop(sessionID)
        },
        'permission.ask': async (_permission, output) => {
          if (output.status === 'ask') await notify('PermissionRequest')
        }
      }
    }
